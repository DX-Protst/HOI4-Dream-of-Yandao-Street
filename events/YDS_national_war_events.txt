			1035 = { add_core_of = ECE transfer_state_to = ECE }

			749 = { add_core_of = ECE transfer_state_to = ECE }

			608 = { add_core_of = EBS transfer_state_to = EBS }
			1039 = { add_core_of = EBS transfer_state_to = EBS }
			328 = { add_core_of = EBS transfer_state_to = EBS }
			609 = { add_core_of = EBS transfer_state_to = EBS }

			745 = { add_core_of = EBS transfer_state_to = EBS }
			761 = { add_core_of = EBS transfer_state_to = EBS }

			# All Top 3 Winners receive Ivory Tower Hegemony (Strong buffs, isolated and cannot ally)
			every_country = {
				limit = {
					has_country_flag = is_gaokao_top3_winner
				}
				leave_faction = yes
				if = {
					limit = { NOT = { has_idea = YDS_ivory_tower_hegemony } }
					add_ideas = YDS_ivory_tower_hegemony
				}
			}

			# Prepare all repeater losers: leave previous factions and gain Repeater Pact with faction permissions
			every_country = {
				limit = {
					has_country_flag = is_gaokao_repeater_loser
				}
				leave_faction = yes
				if = {
					limit = { NOT = { has_idea = YDS_repeater_coalition_pact } }
					add_ideas = YDS_repeater_coalition_pact
				}
			}

			# Pick one repeater loser to establish the Repeater Coalition
			random_country = {
				limit = {
					has_country_flag = is_gaokao_repeater_loser
					exists = yes
				}
				create_faction = FACTION_REPEATER_COALITION
				save_event_target_as = repeater_leader
			}

			# Add all other losers into the Repeater Coalition (Never call leave_faction on leader!)
			every_country = {
				limit = {
					has_country_flag = is_gaokao_repeater_loser
					NOT = { tag = event_target:repeater_leader }
				}
				save_event_target_as = current_coalition_member
				event_target:repeater_leader = {
					add_to_faction = event_target:current_coalition_member
				}
			}

			# Coalition Leader declares war on each of the 3 isolated Winners
			every_country = {
				limit = { has_country_flag = is_gaokao_top3_winner }
				save_event_target_as = current_target_winner
				event_target:repeater_leader = {
					if = {
						limit = { NOT = { has_war_with = event_target:current_target_winner } }
						declare_war_on = { target = event_target:current_target_winner type = annex_everything }
					}
				}
			}

			# ALL OTHER coalition members join the war against each of the 3 Winners
			every_country = {
				limit = {
					has_country_flag = is_gaokao_repeater_loser
					NOT = { tag = event_target:repeater_leader }
				}
				save_event_target_as = current_coalition_member
				every_country = {
					limit = { has_country_flag = is_gaokao_top3_winner }
					save_event_target_as = current_target_winner
					event_target:current_coalition_member = {
						if = {
							limit = { NOT = { has_war_with = event_target:current_target_winner } }
							add_to_war = {
								targeted_alliance = event_target:repeater_leader
								enemy = event_target:current_target_winner
							}
						}
						if = {
							limit = { NOT = { has_war_with = event_target:current_target_winner } }
							declare_war_on = { target = event_target:current_target_winner type = annex_everything }
						}
					}
				}
