# Border War resolution events for PRC expansion decisions
# Pure ASCII comments only

add_namespace = SEA_PRC_border_war_events

# Guyuan Border War: Attacker Victory
country_event = {
	id = SEA_PRC_border_war_events.1
	title = SEA_PRC_border_war_events.1.t
	desc = SEA_PRC_border_war_events.1.d
	picture = GFX_report_event_chinese_officers

	is_triggered_only = yes

	option = {
		name = SEA_PRC_border_war_events.1.a
		if = {
			limit = {
				tag = PRC
			}
			1045 = {
				transfer_state_to = PRC
			}
			add_war_support = 0.05
			army_experience = 10
		}
		if = {
			limit = {
				NOT = { tag = PRC }
			}
			add_war_support = -0.05
		}
	}
}
